>>> MyList = __import__('1-my_list').MyList

# Basic usage
>>> my_list = MyList()
>>> my_list.append(1)
>>> my_list.append(4)
>>> my_list.append(2)
>>> my_list.append(3)
>>> my_list.append(5)
>>> print(my_list)
[1, 4, 2, 3, 5]
>>> my_list.print_sorted()
[1, 2, 3, 4, 5]
>>> print(my_list)
[1, 4, 2, 3, 5]

# Already sorted
>>> my_list2 = MyList()
>>> my_list2.append(1)
>>> my_list2.append(2)
>>> my_list2.append(3)
>>> my_list2.print_sorted()
[1, 2, 3]
>>> print(my_list2)
[1, 2, 3]

# Reverse sorted
>>> my_list3 = MyList()
>>> my_list3.append(5)
>>> my_list3.append(4)
>>> my_list3.append(3)
>>> my_list3.append(2)
>>> my_list3.append(1)
>>> my_list3.print_sorted()
[1, 2, 3, 4, 5]
>>> print(my_list3)
[5, 4, 3, 2, 1]

# One element
>>> my_list4 = MyList()
>>> my_list4.append(7)
>>> my_list4.print_sorted()
[7]

# Empty list
>>> my_list5 = MyList()
>>> my_list5.print_sorted()
[]

# Negative numbers
>>> my_list6 = MyList()
>>> my_list6.append(-3)
>>> my_list6.append(0)
>>> my_list6.append(-1)
>>> my_list6.append(2)
>>> my_list6.print_sorted()
[-3, -1, 0, 2]
>>> print(my_list6)
[-3, 0, -1, 2]

# Duplicate numbers
>>> my_list7 = MyList()
>>> my_list7.append(3)
>>> my_list7.append(1)
>>> my_list7.append(3)
>>> my_list7.append(1)
>>> my_list7.print_sorted()
[1, 1, 3, 3]
>>> print(my_list7)
[3, 1, 3, 1]
